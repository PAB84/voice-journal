AlarmManager, Notifications & Room Database

1. Android AlarmManager & Notification Broadcast 
class ReminderReceiver : BroadcastReceiver() {
    override fun onReceive(context: Context, intent: Intent) {
        val title = intent.getStringExtra("EXTRA_TITLE") ?: "Task Reminder"
        
        // 1. Post Android System Notification
        val notification = NotificationCompat.Builder(context, "reminders_channel")
            .setSmallIcon(R.drawable.ic_notification)
            .setContentTitle("⏰ Reminder Alert")
            .setContentText(title)
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .build()
        NotificationManagerCompat.from(context).notify(System.currentTimeMillis().toInt(), notification)
        
        // 2. Play Audio Chime & Speak out loud
        val tts = TextToSpeech(context) { status ->
            if (status == TextToSpeech.SUCCESS) {
                tts.speak("Reminder alert: $title", TextToSpeech.QUEUE_FLUSH, null, null)
            }
        }
    }
}
2. Scheduling Exact Alarms 
fun scheduleReminder(context: Context, title: String, triggerAtMillis: Long) {
    val alarmManager = context.getSystemService(Context.ALARM_SERVICE) as AlarmManager
    val intent = Intent(context, ReminderReceiver::class.java).apply {
        putExtra("EXTRA_TITLE", title)
    }
    val pendingIntent = PendingIntent.getBroadcast(
        context, title.hashCode(), intent, 
        PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
    )
    alarmManager.setExactAndAllowWhileIdle(
        AlarmManager.RTC_WAKEUP, triggerAtMillis, pendingIntent
    )
}
